class Solution {
    public int countBattleships(char[][] board) {
        int shipCount = 0;
        boolean[][] visited = new boolean[board.length][board[0].length];

        for(short y = 0; y < board.length; y++) {
            for(short x = 0; x < board[0].length; x++) {
                if(board[y][x] == 'X' && visited[y][x] == false) {
                    shipCount++;
                    mapShip(board, visited, y, x);
                }
            }
        }

        return shipCount;
    }

    public void mapShip(char[][] board, boolean[][] visited, int y, int x) {
        if(x < 0 || x == board[0].length || y < 0 || y == board.length || visited[y][x] == true ||
        board[y][x] == '.') {
            return;
        }

        visited[y][x] = true;

        mapShip(board,visited, y - 1, x);
        mapShip(board,visited, y + 1, x);
        mapShip(board,visited, y, x - 1);
        mapShip(board,visited, y, x + 1);
    }
}
