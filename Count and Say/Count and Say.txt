class Solution {
    public String countAndSay(int n) {
        if(n == 1) {
            return "1";
        }

        if(n == 2) {
            return "11";
        }

        return countAndSay(n - 2, "11");
    }

    public String countAndSay(int n, String s) {
        if(n == 0){
            return s;
        }

        StringBuilder sb = new StringBuilder();
        int count = 1;
        for(int i = 0; i < s.length() - 1; i++) {
            char curr = s.charAt(i);
            char next = s.charAt(i + 1);

            if(curr == next) {
                count++;

                if(i == s.length() - 2){
                    sb.append(count).append(curr);
                }
                continue;
            } 

            if(curr != next) {
                sb.append(count).append(curr);
                count = 1;

                if(i == s.length() - 2) {
                    sb.append(count).append(next);
                }
            }
        }

        return countAndSay(n - 1, sb.toString());
    }
}
