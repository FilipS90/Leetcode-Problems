class Solution {
    public int maxAreaOfIsland(int[][] grid) {
        int maxIsland = 0;
        boolean[][] visited = new boolean[grid.length][grid[0].length];

        for(int y = 0; y < grid.length; y++) {
            for(int x = 0; x < grid[0].length; x++) {
                if(grid[y][x] == 1 && visited[y][x] == false) {
                    AtomicInteger islandSize = new AtomicInteger(0);
                    mapIsland(grid, visited, y, x, islandSize);
                    maxIsland = maxIsland < islandSize.intValue() ? islandSize.intValue() : maxIsland;
                }
            }
        }

        return maxIsland;
    }

    public void mapIsland(int[][] grid, boolean[][] visited, int y, int x, AtomicInteger count) {
        if(x < 0 || x == grid[0].length || y < 0 || y == grid.length || visited[y][x] == true ||
        grid[y][x] == 0) {
            return;
        }

        count.addAndGet(1);
        visited[y][x] = true;

        mapIsland(grid,visited, y - 1, x, count);
        mapIsland(grid,visited, y + 1, x, count);
        mapIsland(grid,visited, y, x - 1, count);
        mapIsland(grid,visited, y, x + 1, count);
    }
}
