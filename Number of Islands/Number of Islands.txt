class Solution {
    public int numIslands(char[][] grid) {
        boolean [][] visited = new boolean [grid.length][grid[0].length];
        int islandCount = 0;

        for(int y = 0; y < grid.length; y++) {
            for(int x = 0; x < grid[0].length; x++) {
                if(grid[y][x] == '1' && visited[y][x] == false){
                    islandCount++;
                    mapIsland(visited, y, x, grid);
                }
            }
        }

        return islandCount;
    }

    public void mapIsland(boolean[][] visited, int y, int x, char[][] grid) {
        if(y < 0 || y == grid.length || x < 0 || x == grid[0].length ||
        visited[y][x] == true || grid[y][x] == '0'){
            return;
        }

        visited[y][x] = true;

        mapIsland(visited, y + 1, x, grid);
        mapIsland(visited, y, x + 1, grid);
        mapIsland(visited, y - 1, x, grid);
        mapIsland(visited, y, x - 1, grid);
    }
}
