class Solution {
    public ListNode partition(ListNode head, int x) {
        if(head == null)
            return head;

        List<ListNode> less = new ArrayList();
        List<ListNode> greaterOrEqual = new ArrayList();
        ListNode tmp = head;

        while(tmp != null) {
            if(tmp.val < x) {
                less.add(tmp);
            } else {
                greaterOrEqual.add(tmp);
            }

            tmp = tmp.next;
        }

        if(less.size() == 0)
            return head;

        ListNode newHead = less.get(0);
        tmp = newHead;

        for(int i = 1; i < less.size(); i++) {
            tmp.next = less.get(i);
            tmp = tmp.next;
        }

        for(int i = 0; i < greaterOrEqual.size(); i++) {
            tmp.next = greaterOrEqual.get(i);
            tmp = tmp.next;
        }

        tmp.next = null;

        return newHead;
    }
}
